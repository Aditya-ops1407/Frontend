This is how you write a multiline code:

        
            import os

            #Write a file
            st = ("I am a python developer\nI want to be a programmer\nI am getting better everyday")
            f = open("myfile.txt","w") # "w" to specify write mode
            f.write(st)
            f.close()
            
            #Read a File
            f = open("myfile.txt")
            data = f.read()
            print(data)
            f.close()
            
            #Readlines [Gives Output in form of list]
            f = open("myfile.txt")
            lines = f.readlines()
            print(lines , type(lines))
            f.close()
